# Стратегия глубокой обороны для СССР (до и во время войны с Германией)
smart_ai_soviet_deep_defense = {
    allowed = { always = yes }
    enable = {
        is_ai = yes
        tag = SOV
        has_war_with = GER   # Активируется только с началом войны
        date < 1942.1.1
    }
    abort_when_not_enabled = yes

    # Запрещаем атаки и заставляем держать позиции
    ai_strategy = {
        type = front_control
        priority = 200
        ordertype = front
        execution_type = careful   # Осторожный режим
        manual_attack = no         # Запрет на мелкие атаки
        execute_order = yes        # Выполнять приказы на перемещение (но не атаку)
    }

    # Усиливаем гарнизоны в тылу (дивизии в городах окапываются)
    ai_strategy = { type = role_ratio  value = 200 }

    # Приоритет обороны ключевых регионов
    ai_strategy = { type = area_priority id = europe value = 300 }
    ai_strategy = { type = area_priority id = defend_moscow_zone value = 250 }
}
