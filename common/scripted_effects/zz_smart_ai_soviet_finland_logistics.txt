    ROOT = {
        if = {
            limit = {
                tag = SOV
                is_ai = yes
                NOT = { has_war_with = FIN }
                date > 1938.1.1
                date < 1940.1.1
            }

            every_owned_state = {
                limit = {
                    OR = {
                        state = 219 # Москва
                        state = 263 # Новгород
                        state = 195 # Ленинград
                        state = 213 # Восточная Карелия
                        state = 214 # Архангельск
                        state = 215 # Онега
                    }
                }

                save_event_target_as = current_state

                # Слоты добавляем только один раз
                if = {
                    limit = { NOT = { has_state_flag = sov_fin_logistics_slots_added } }
                    add_extra_state_shared_building_slots = 5
                    set_state_flag = sov_fin_logistics_slots_added
                }

                # 1. Инфраструктура до 5
                if = {
                    limit = { infrastructure < 5 }
                    add_building_construction = {
                        type = infrastructure
                        level = 5
                        instant_build = yes
                    }
                }

                # 2. Железная дорога (закомментировано, нужен явный path)
                # if = {
                #     limit = {
                #         NOT = {

                #             }
                #         }
                #     }
                #     build_railway = { path = { ... } }
                # }

                # 3. Узел снабжения: строим ТОЛЬКО если его нет и флаг не установлен
                if = {
                    limit = {
                        NOT = { has_state_flag = sov_fin_supply_node_built }
                        NOT = {
                            any_province_building_level = {
                                building = supply_node
                                level > 0
                            }
                        }
                    }
                    construct_building_in_random_province = {
                        supply_node = 1
                    }
                    set_state_flag = sov_fin_supply_node_built
                }
            }
        }
    }
}
