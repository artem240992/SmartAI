# Германия и её союзники: «Русская зима»
country_event = {
    id = zz_smart_ai_seasonal.2
    title = zz_smart_ai_seasonal.2.t
    desc = zz_smart_ai_seasonal.2.d
    picture = GFX_report_event_german_winter
    is_triggered_only = yes
    trigger = {
        has_war_with = SOV
        OR = {
            tag = GER
            is_in_faction_with = GER
        }
    }
    immediate = {
        # Дебафф получает та страна, на которую сработало событие (Германия или союзник)
        add_dynamic_modifier = { modifier = german_winter_debuff days = 90 }
        
        # Синхронизация контрнаступления: запускаем ТОЛЬКО если событие получила Германия
        if = {
            limit = {
                tag = GER
                SOV = { is_ai = yes }
                NOT = { SOV = { has_idea = soviet_winter_counterattack_idea } }
            }
            SOV = {
                add_timed_idea = { idea = soviet_winter_counterattack_idea days = 100 }
            }
            soviet_winter_counterattack_effect = yes
        }
    }
    option = { name = zz_smart_ai_seasonal.2.a }
}
